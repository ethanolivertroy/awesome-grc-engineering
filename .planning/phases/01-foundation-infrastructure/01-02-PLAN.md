```yaml
    runs-on: ubuntu-latest

      - name: Checkout

      - name: Run awesome-lint

```

**link-check.yml** (from STACK.md Section 4.3):
```yaml
name: Link Check

on:
  schedule:
    - cron: '0 0 * * 0'
  pull_request:
    branches: [main]
  workflow_dispatch:

jobs:
  linkChecker:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Restore lychee cache
        uses: actions/cache@v4
        with:
          path: .lycheecache
          key: cache-lychee-${{ github.sha }}
          restore-keys: cache-lychee-

      - name: Link Checker
        uses: lycheeverse/lychee-action@v2
        with:
          args: '--cache --max-cache-age 1d --verbose --no-progress .'
          fail: true
          jobSummary: true

      - name: Create Issue From File
        if: failure() && github.event_name == 'schedule'
        uses: peter-evans/create-issue-from-file@v5
        with:
          title: 'Broken Links Detected'
          content-filepath: ./lychee/out.md
          labels: |
            automated
            broken-links
```

Key: Weekly schedule, caching for rate limits, auto-issue on failure during scheduled runs only.

**toc.yml** (from STACK.md Section 4.4):
```yaml
name: TOC Generator

on:
  push:
    branches: [main]
    paths:
      - 'README.md'

jobs:
  generateTOC:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: '20'

      - name: Generate TOC
        run: |
          npm install -g doctoc
          doctoc README.md --title '## Contents'

      - name: Commit TOC
        uses: stefanzweifel/git-auto-commit-action@v5
        with:
          commit_message: 'docs: update table of contents'
          file_pattern: 'README.md'
```

Key: Only triggers on README.md changes, uses --title to match awesome convention.
